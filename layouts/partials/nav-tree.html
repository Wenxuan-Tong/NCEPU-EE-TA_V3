"max" 5 "showRoot" false) }} ========================================================= */}} {{- $ctx := .ctx -}} {{- $max := .max | default 5 -}} {{- $showRoot := .showRoot | default false -}} {{/* 锁定当前页面所属的顶层区段 */}} {{- $root := or $ctx.FirstSection $ctx.CurrentSection -}} {{- if not $root -}} {{- with $ctx.Section -}} {{- $root = site.GetPage (printf "section/%s" .) -}} {{- end -}} {{- end -}} {{- if $root -}} {{/* 统一收集子项（Sections + RegularPages）并按 Weight/Title 排序 */}} {{- $kids := slice -}} {{- range $root.Sections }}{{ $kids = $kids | append . }}{{ end -}} {{- range $root.RegularPages }}{{ $kids = $kids | append . }}{{ end -}} {{- $kids = sort $kids "Weight" "Title" -}}
{{- if $showRoot -}} {{ template "hbnode" (dict "node" $root "ctx" $ctx "level" 1 "max" $max) }} {{- else -}} {{- range $kids -}} {{ template "hbnode" (dict "node" . "ctx" $ctx "level" 1 "max" $max) }} {{- end -}} {{- end -}}
{{- else -}}
{{- end -}} {{/* =============== 递归节点渲染 =============== */}} {{- define "hbnode" -}} {{- $node := .node -}} {{- $ctx := .ctx -}} {{- $level := .level -}} {{- $max := .max -}} {{- $isActive := eq $node.RelPermalink $ctx.RelPermalink -}} {{- $isOpen := or $isActive ($node.IsAncestor $ctx) -}} {{- $hasChildren := or (gt (len $node.Sections) 0) (gt (len $node.RegularPages) 0) -}} {{- $hasMore := and $hasChildren (lt $level $max) -}} {{- $title := $node.LinkTitle | default $node.Title -}} {{- $levelClass := printf "level-%d" $level -}} {{- if and $hasMore (or (eq $node.Kind "section") (eq $node.Kind "branch")) -}}
{{ $title }} {{- $secs := sort $node.Sections "Weight" "Title" -}} {{- range $secs -}} {{ template "hbnode" (dict "node" . "ctx" $ctx "level" (add $level 1) "max" $max) }} {{- end -}} {{- $pages := sort $node.RegularPages "Weight" "Title" -}} {{- range $pages -}} {{ template "hbnode" (dict "node" . "ctx" $ctx "level" (add $level 1) "max" $max) }} {{- end -}}
{{- else -}}

"mb-3" -}} {{- else -}} {{/* center（默认） */}} {{- $wrapCls = printf "%s text-center" $wrapCls -}} {{- $cls = $cls | append "d-block" "mx-auto" -}} {{- end -}} {{- $imgCls := delimit $cls " " -}} {{/*

{{- /* Hugo Shortcode: hb-gallery 用法：在页面里写 {{< hb-gallery >}} 图片来源优先顺序： 1) front matter 的 resources（可带 title/caption） 2) 页面包内的 images/* 资源 3) 页面下所有 image 资源 */ -}}
{{- $imgs := slice -}} {{/* 1) 前置 resources（可带元数据） */}} {{- with .Page.Resources }} {{- range . }} {{- if eq .ResourceType "image" }} {{- $imgs = $imgs | append . }} {{- end }} {{- end }} {{- end }} {{/* 2) 若无 resources，则匹配 images/* */}} {{- if eq (len $imgs) 0 -}} {{- $imgs = .Page.Resources.Match "images/*" -}} {{- end }} {{/* 3) 兜底：页面所有 image 资源 */}} {{- if eq (len $imgs) 0 -}} {{- $imgs = .Page.Resources.ByType "image" -}} {{- end }} {{- range $i, $img := $imgs }} {{- $title := or $img.Title $img.Params.title (replace $img.Name "-" " ") -}} {{- $cap := or $img.Params.caption $title -}} {{- $thumb := $img.Fit "800x800 webp" -}} {{- end }}
